3. Create `src/lib/opencc.ts` — OpenCC conversion with custom dictionary per D-06 (static, loaded at build time):
   ```typescript
   import { Converter } from 'opencc-js';
   import metaphysicsDict from '@/dictionaries/metaphysics-zh-hans.json';

   // Pre-built converters (D-06: static, loaded at build time)
   const toTraditional = Converter({ from: 'cn', to: 't' });
   const toSimplified = Converter({ from: 't', to: 'cn' });

   /**
    * Convert text to Traditional Chinese with metaphysics term protection.
    * First applies OpenCC generic conversion, then overwrites any metaphysics
    * terms that were incorrectly converted.
    */
   export function convertToTraditional(text: string): string {
     // Step 1: Apply standard OpenCC conversion
     let result = toTraditional(text);
     // Step 2: Fix metaphysics terms that OpenCC gets wrong
     // The dictionary maps simplified→traditional, so we need to check
     // if any simplified term appears in the original and its traditional
     // form was incorrectly converted
     for (const [simplified, traditional] of Object.entries(metaphysicsDict)) {
       // If the original text contained the simplified term,
       // ensure the result has the correct traditional form
       if (text.includes(simplified)) {
         // OpenCC may have converted it to something else — fix it
         const wrongConversion = toTraditional(simplified);
         if (wrongConversion !== traditional) {
           result = result.replace(new RegExp(wrongConversion, 'g'), traditional);
         }
       }
     }
     return result;
   }

   /**
    * Convert text to Simplified Chinese with metaphysics term protection.
    */
   export function convertToSimplified(text: string): string {
     let result = toSimplified(text);
     // Build reverse dictionary (traditional→simplified)
     for (const [simplified, traditional] of Object.entries(metaphysicsDict)) {
       if (text.includes(traditional)) {
         const wrongConversion = toSimplified(traditional);
         if (wrongConversion !== simplified) {
           result = result.replace(new RegExp(wrongConversion, 'g'), simplified);
         }
       }
     }
     return result;
   }

   /**
    * Convert metaphysics-specific terms between Traditional and Simplified.
    * Use this for raw term conversion (not full text).
    */
   export function convertMetaphysics(term: string, targetLocale: 'zh-hant' | 'zh-hans'): string {
     if (targetLocale === 'zh-hant') {
       return (metaphysicsDict as Record<string, string>)[term] || term;
     }
     // Reverse lookup for zh-hans
     const reverse = Object.fromEntries(
       Object.entries(metaphysicsDict).map(([k, v]) => [v, k])
     );
     return (reverse as Record<string, string>)[term] || term;
   }
   ```
